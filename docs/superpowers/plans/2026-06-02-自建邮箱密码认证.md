## Task 2：改造 api/auth.ts（注册/登录 hooks）

**Files:** Modify `image-web/src/api/auth.ts`

- [ ] **Step 1**：删 `useLogin`（OAuth）；新增 `useRegister`/`useLoginPwd`，均写 token 后由 ProtectedRoute 拉 /me：
```ts
export function useRegister() {
  const setToken = useAuthStore((s) => s.setToken)
  return useMutation({
    mutationFn: async (v: { email: string; password: string; name: string }) => {
      const { data, error } = await api.POST('/auth/register', { body: v })
      if (error || !data) throw new Error(errorMessage(error, '注册失败'))
      return data
    },
    onSuccess: (d) => setToken(d.jwt),
  })
}
export function useLoginPwd() {
  const setToken = useAuthStore((s) => s.setToken)
  return useMutation({
    mutationFn: async (v: { email: string; password: string }) => {
      const { data, error } = await api.POST('/auth/login', { body: v })
      if (error || !data) throw new Error(errorMessage(error, '登录失败'))
      return data
    },
    onSuccess: (d) => setToken(d.jwt),
  })
}
```
- [ ] **Step 2** 验证：`npm run typecheck`（LoginPage 仍引用旧 useLogin→红，Task 3 修）。
- [ ] **Step 3** Commit：`git add image-web/src/api/auth.ts && git commit -m "feat(web): auth hooks 改邮箱密码注册/登录"`
